#include <stdio.h>
#include <pthread.h>

void *threadFunction(void *arg)
{
    printf("Thread is executing...\n");
    return NULL;
}

int main()
{
    pthread_t tid;

    pthread_create(&tid, NULL, threadFunction, NULL);

    pthread_join(tid, NULL);

    printf("Main Thread Finished\n");

    return 0;
}
