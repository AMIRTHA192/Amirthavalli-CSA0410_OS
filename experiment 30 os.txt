#include <stdio.h>

struct Block
{
    int blockNumber;
    int next;
};

int main()
{
    struct Block blocks[50];
    int n, i, record, current;

    printf("Enter number of blocks in the file: ");
    scanf("%d", &n);

    printf("Enter block numbers:\n");

    for (i = 0; i < n; i++)
    {
        scanf("%d", &blocks[i].blockNumber);

        if (i == n - 1)
            blocks[i].next = -1;
        else
            blocks[i].next = i + 1;
    }

    printf("\nLinked File Allocation\n");

    printf("Start Block = %d\n", blocks[0].blockNumber);
    printf("End Block   = %d\n", blocks[n - 1].blockNumber);

    printf("\nBlock Chain:\n");

    current = 0;

    while (current != -1)
    {
        printf("%d", blocks[current].blockNumber);

        if (blocks[current].next != -1)
            printf(" -> ");

        current = blocks[current].next;
    }

    printf("\n");

    printf("\nEnter record number to access (1-%d): ", n);
    scanf("%d", &record);

    if (record < 1 || record > n)
    {
        printf("Invalid record number.\n");
        return 0;
    }

    current = 0;

    for (i = 1; i < record; i++)
        current = blocks[current].next;

    printf("Record %d is stored in block %d.\n",
           record, blocks[current].blockNumber);

    return 0;
}
