#include <stdio.h>

int main()
{
    int indexBlock, n, blocks[50];
    int i, record;

    printf("Enter index block number: ");
    scanf("%d", &indexBlock);

    printf("Enter number of file blocks: ");
    scanf("%d", &n);

    printf("Enter the block numbers:\n");

    for (i = 0; i < n; i++)
    {
        scanf("%d", &blocks[i]);
    }

    printf("\nFile Allocation Table\n");
    printf("Index Block: %d\n", indexBlock);

    printf("Index Block -> ");

    for (i = 0; i < n; i++)
    {
        printf("%d", blocks[i]);

        if (i != n - 1)
            printf(" -> ");
    }

    printf("\n");

    printf("\nEnter record number to access (1-%d): ", n);
    scanf("%d", &record);

    if (record >= 1 && record <= n)
    {
        printf("Record %d is stored in block %d.\n",
               record, blocks[record - 1]);
    }
    else
    {
        printf("Invalid record number.\n");
    }

    return 0;
}
