
#include <stdio.h>
#include <stdlib.h>

int main()
{
    int a[50], n, head, disk, direction;
    int i, j, temp, total = 0;

    printf("Enter number of requests: ");
    scanf("%d", &n);

    printf("Enter request queue:\n");

    for(i = 0; i < n; i++)
        scanf("%d", &a[i]);

    printf("Enter initial head position: ");
    scanf("%d", &head);

    printf("Enter disk size: ");
    scanf("%d", &disk);

    printf("Enter direction (1=Right, 0=Left): ");
    scanf("%d", &direction);

    /* Sort requests */
    for(i = 0; i < n - 1; i++)
    {
        for(j = i + 1; j < n; j++)
        {
            if(a[i] > a[j])
            {
                temp = a[i];
                a[i] = a[j];
                a[j] = temp;
            }
        }
    }

    printf("\nSeek sequence: ");

    if(direction == 1)
    {
        /* Move right */
        for(i = 0; i < n; i++)
        {
            if(a[i] >= head)
            {
                total += abs(head - a[i]);
                head = a[i];
                printf("%d ", head);
            }
        }

        /* Go to end */
        total += abs(head - (disk - 1));
        head = disk - 1;

        /* Jump to beginning */
        total += disk - 1;
        head = 0;

        /* Continue right */
        for(i = 0; i < n; i++)
        {
            if(a[i] > 0)
            {
                total += abs(head - a[i]);
                head = a[i];
                printf("%d ", head);
            }
        }
    }
    else
    {
        /* Move left */
        for(i = n - 1; i >= 0; i--)
        {
            if(a[i] <= head)
            {
                total += abs(head - a[i]);
                head = a[i];
                printf("%d ", head);
            }
        }

        /* Go to beginning */
        total += head;
        head = 0;

        /* Jump to end */
        total += disk - 1;
        head = disk - 1;

        /* Continue left */
        for(i = n - 1; i >= 0; i--)
        {
            if(a[i] < disk - 1)
            {
                total += abs(head - a[i]);
                head = a[i];
                printf("%d ", head);
            }
        }
    }

    printf("\nTotal head movement = %d\n", total);

    return 0;
}
